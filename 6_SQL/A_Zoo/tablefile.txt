drop table ZooAnimal
drop table Animal
drop table Zoo

create table Animal
(
    [Id]   int identity primary key,
    [Name] nvarchar(50)
);

create table Zoo
(
    [Id]       int identity primary key,
    [Location] nvarchar(50)
);

create table ZooAnimal
(
    [Id]       int identity primary key,

    [ZooId]    int not null
        constraint Zoo___fk references Zoo on delete cascade,

    [AnimalId] int not null
        constraint Animal___fk references Animal on delete cascade
);
